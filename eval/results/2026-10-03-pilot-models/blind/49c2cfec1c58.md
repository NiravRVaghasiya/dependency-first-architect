# **Multi-Tenant SaaS Billing System Architecture**

## **1. System Overview**

A multi-tenant billing system serves multiple customers (tenants) from a shared infrastructure while maintaining complete data isolation and customization per tenant.

### **Key Requirements:**
- Multi-tenancy with data isolation
- Subscription management
- Usage-based billing
- Invoice generation
- Payment processing
- Compliance & audit trails
- Real-time metering
- Analytics & reporting

```python
class PaymentService:
    
    def process_payment(tenant_id, invoice_id, payment_method_id=None):
        """Process payment for invoice"""
        invoice = validate_tenant_invoice(tenant_id, invoice_id)
        
        if invoice['status'] == 'paid':
            raise AlreadyPaidError()
        
        payment_method = get_payment_method(tenant_id, payment_method_id)
        
        # Charge via payment processor
        processor = PaymentProcessor.get(payment_method['type'])
        
        try:
            result = processor.charge(
                amount=invoice['total_amount'],
                currency='USD',
                customer_token=payment_method['provider_id'],
                idempotency_key=str(invoice['invoice_id'])
            )
            
            payment = {
                'payment_id': uuid(),
                'tenant_id': tenant_id,
                'invoice_id': invoice_id,
                'account_id': invoice['account_id'],
                'amount': invoice['total_amount'],
                'status': 'completed',
                'external_transaction_id': result['transaction_id'],
                'processed_at': now()
            }
            
            db.insert('payments', payment)
            db.update('invoices', invoice_id, {
                'status': 'paid',
                'paid_at': now()
            })
            
            event_bus.publish('payment.processed', payment)
            
        except PaymentFailedError as e:
            payment = {
                'payment_id': uuid(),
                'tenant_id': tenant_id,
                'invoice_id': invoice_id,
                'account_id': invoice['account_id'],
                'amount': invoice['total_amount'],
                'status': 'failed',
                'error_message': str(e),
                'created_at': now()
            }
            
            db.insert('payments', payment)
            event_bus.publish('payment.failed', payment)
            raise
    
    def store_payment_method(tenant_id, account_id, card_data):
        """Tokenize and store payment method securely"""
        validate_tenant_account(tenant_id, account_id)
        
        processor = PaymentProcessor.get('credit_card')
        token = processor.tokenize(card_data)
        
        payment_method = {
            'payment_method_id': uuid(),
            'tenant_id': tenant_id,
            'account_id': account_id,
            'type': 'credit_card',
            'provider_id': token,
            'is_default': False,
            'last_four': card_data['number'][-4:],
            'expiry_month': card_data['exp_month'],
            'expiry_year': card_data['exp_year']
        }
        
        db.insert('payment_methods', payment_method)
        return payment_method
```

## **6. Proration Logic**

```python
def calculate_proration(old_subscription, new_plan):
    """
    Handle mid-cycle plan changes
    """
    current_period_end = old_subscription['current_period_end']
    days_remaining = (current_period_end - today()).days
    
    old_plan = db.get_plan(old_subscription['plan_id'])
    new_plan = db.get_plan(new_plan['plan_id'])
    
    daily_old = old_plan['base_price'] / days_in_month
    daily_new = new_plan['base_price'] / days_in_month
    
    credit = daily_old * days_remaining
    charge = daily_new * days_remaining
    
    return {
        'credit': credit,
        'charge': charge,
        'net_adjustment': charge - credit
    }
```

```python
def tenant_context_middleware(request):
    """Extract and validate tenant context"""
    token = extract_bearer_token(request)
    claims = jwt.decode(token, SECRET_KEY)
    
    tenant_id = claims.get('tenant_id')
    request_tenant_id = request.headers.get('X-Tenant-ID')
    
    # Token tenant_id must match request
    if tenant_id != request_tenant_id:
        raise UnauthorizedError()
    
    # Inject into request context
    request.tenant_id = tenant_id
    request.user_id = claims.get('user_id')
    request.account_id = claims.get('account_id')
```
