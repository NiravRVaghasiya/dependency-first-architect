### A. Billing Service (Node.js/TypeScript)

```typescript
import { EventEmitter } from 'events';
import { Pool } from 'pg';
import * as stripe from 'stripe';

interface SubscriptionPlan {
  id: string;
  tenantId: string;
  name: string;
  billingPeriod: 'monthly' | 'annual';
  amountCents: number;
  currency: string;
  features: Record<string, unknown>;
}

interface Subscription {
  id: string;
  billingAccountId: string;
  planId: string;
  status: 'active' | 'paused' | 'cancelled' | 'past_due';
  currentPeriodStart: Date;
  currentPeriodEnd: Date;
  quantity: number;
}

class BillingService {
  private db: Pool;
  private stripeClient: stripe.Stripe;
  private eventEmitter: EventEmitter;

  constructor(db: Pool, stripeKey: string, eventEmitter: EventEmitter) {
    this.db = db;
    this.stripeClient = new stripe.Stripe(stripeKey, { apiVersion: '2023-10-16' });
    this.eventEmitter = eventEmitter;
  }

  /**
   * Create a subscription for a billing account
   */
  async createSubscription(
    billingAccountId: string,
    planId: string,
    quantity: number = 1
  ): Promise<Subscription> {
    const client = await this.db.connect();
    try {
      await client.query('BEGIN');

      // Fetch plan details
      const planResult = await client.query(
        'SELECT * FROM pricing_plans WHERE id = $1',
        [planId]
      );
      if (planResult.rows.length === 0) {
        throw new Error('Plan not found');
      }
      const plan: SubscriptionPlan = planResult.rows[0];

      // Create subscription record
      const subscriptionId = this.generateId();
      const now = new Date();
      const periodEnd = this.calculatePeriodEnd(now, plan.billingPeriod);

      await client.query(
        `INSERT INTO subscriptions 
         (id, billing_account_id, plan_id, status, current_period_start, current_period_end, quantity) 
         VALUES ($1, $2, $3, $4, $5, $6, $7)`,
        [subscriptionId, billingAccountId, planId, 'active', now, periodEnd, quantity]
      );

      // Emit event for invoice generation
      this.eventEmitter.emit('subscription.created', {
        subscriptionId,
        billingAccountId,
        planId,
        amount: plan.amountCents * quantity,
      });

      await client.query('COMMIT');

      return {
        id: subscriptionId,
        billingAccountId,
        planId,
        status: 'active',
        currentPeriodStart: now,
        currentPeriodEnd: periodEnd,
        quantity,
      };
    } catch (error) {
      await client.query('ROLLBACK');
      throw error;
    } finally {
      client.release();
    }
  }

  /**
   * Calculate invoice charges for a subscription
   */
  async calculateInvoiceAmount(subscriptionId: string): Promise<number> {
    const query = `
      SELECT 
        s.quantity * p.amount_cents as base_charge,
        COALESCE(SUM(CASE 
          WHEN ur.timestamp >= s.current_period_start 
            AND ur.timestamp < s.current_period_end 
          THEN pt.price_per_unit_cents * ur.quantity 
          ELSE 0 
        END), 0) as usage_charges
      FROM subscriptions s
      JOIN pricing_plans p ON s.plan_id = p.id
      LEFT JOIN usage_records ur ON s.id = ur.subscription_id
      LEFT JOIN plan_tiers pt ON p.id = pt.plan_id
      WHERE s.id = $1
      GROUP BY s.quantity, p.amount_cents
    `;

    const result = await this.db.query(query, [subscriptionId]);
    const row = result.rows[0];
    return (row?.base_charge || 0) + (row?.usage_charges || 0);
  }

  /**
   * Process scheduled billing (daily/hourly via cron)
   */
  async processBillingCycle(): Promise<void> {
    const query = `
      SELECT DISTINCT s.id, s.billing_account_id, s.plan_id, s.quantity
      FROM subscriptions s
      WHERE s.status = 'active'
        AND s.current_period_end = CURRENT_DATE
      LIMIT 1000
    `;

    const result = await this.db.query(query);

    for (const row of result.rows) {
      try {
        await this.createInvoiceForSubscription(
          row.id,
          row.billing_account_id,
          row.plan_id,
          row.quantity
        );
      } catch (error) {
        console.error(`Failed to process subscription ${row.id}:`, error);
        this.eventEmitter.emit('billing.error', {
          subscriptionId: row.id,
          error: error instanceof Error ? error.message : 'Unknown error',
        });
      }
    }
  }

  /**
   * Create invoice for subscription
   */
  private async createInvoiceForSubscription(
    subscriptionId: string,
    billingAccountId: string,
    planId: string,
    quantity: number
  ): Promise<void> {
    const client = await this.db.connect();
    try {
      await client.query('BEGIN');

      const invoiceId = this.generateId();
      const invoiceNumber = await this.generateInvoiceNumber(billingAccountId);
      const amountCents = await this.calculateInvoiceAmount(subscriptionId);

      // Create invoice
      await client.query(
        `INSERT INTO invoices 
         (id, billing_account_id, invoice_number, status, amount_cents, issue_date, due_date)
         VALUES ($1, $2, $3, $4, $5, CURRENT_DATE, CURRENT_DATE + INTERVAL '30 days')`,
        [invoiceId, billingAccountId, invoiceNumber, 'finalized', amountCents]
      );

      // Create line items
      await client.query(
        `INSERT INTO invoice_line_items 
         (id, invoice_id, description, quantity, unit_price_cents, amount_cents, type)
         VALUES ($1, $2, $3, $4, $5, $6, $7)`,
        [
          this.generateId(),
          invoiceId,
          'Subscription charge',
          quantity,
          Math.floor(amountCents / quantity),
          amountCents,
          'subscription',
        ]
      );

      // Update subscription period
      await client.query(
        `UPDATE subscriptions 
         SET current_period_start = current_period_end,
             current_period_end = current_period_end + INTERVAL '1 month'
         WHERE id = $1`,
        [subscriptionId]
      );

      await client.query('COMMIT');

      // Emit invoice created event
      this.eventEmitter.emit('invoice.created', {
        invoiceId,
        billingAccountId,
        amount: amountCents,
      });
    } catch (error) {
      await client.query('ROLLBACK');
      throw error;
    } finally {
      client.release();
    }
  }

  /**
   * Handle payment failures and retries
   */
  async retryFailedPayments(): Promise<void> {
    const query = `
      SELECT * FROM transactions
      WHERE status = 'failed'
        AND attempt_count < 3
        AND next_retry_at <= NOW()
      ORDER BY created_at ASC
      LIMIT 100
    `;

    const result = await this.db.query(query);

    for (const transaction of result.rows) {
      await this.retryTransaction(transaction.id);
    }
  }

  private async retryTransaction(transactionId: string): Promise<void> {
    // Implementation for retrying failed transactions
  }

  private calculatePeriodEnd(startDate: Date, period: 'monthly' | 'annual'): Date {
    const end = new Date(startDate);
    if (period === 'monthly') {
      end.setMonth(end.getMonth() + 1);
    } else {
      end.setFullYear(end.getFullYear() + 1);
    }
    return end;
  }

  private generateId(): string {
    return `${Date.now()}-${Math.random().toString(36).substring(2, 11)}`;
  }

  private async generateInvoiceNumber(billingAccountId: string): Promise<string> {
    const result = await this.db.query(
      `SELECT COUNT(*) as count FROM invoices WHERE billing_account_id = $1`,
      [billingAccountId]
    );
    return `INV-${billingAccountId.substring(0, 8)}-${result.rows[0].count + 1}`;
  }
}

export default BillingService;
```

### B. Metering/Usage Service

```typescript
import { Pool } from 'pg';
import { Redis } from 'ioredis';

interface UsageRecord {
  subscriptionId: string;
  metricName: string;
  quantity: number;
  timestamp: Date;
  idempotencyKey?: string;
}

class MeteringService {
  private db: Pool;
  private cache: Redis;

  constructor(db: Pool, cache: Redis) {
    this.db = db;
    this.cache = cache;
  }

  /**
   * Record usage (idempotent via idempotency key)
   */
  async recordUsage(usage: UsageRecord): Promise<void> {
    const idempotencyKey = usage.idempotencyKey || this.generateIdempotencyKey(usage);

    // Check cache for duplicate
    const cached = await this.cache.get(`usage:${idempotencyKey}`);
    if (cached) {
      return; // Duplicate request
    }

    try {
      await this.db.query(
        `INSERT INTO usage_records 
         (subscription_id, metric_name, quantity, timestamp, idempotency_key)
         VALUES ($1, $2, $3, $4, $5)
         ON CONFLICT (idempotency_key) DO NOTHING`,
        [usage.subscriptionId, usage.metricName, usage.quantity, usage.timestamp, idempotencyKey]
      );

      // Cache for 24 hours
      await this.cache.setex(`usage:${idempotencyKey}`, 86400, '1');
    } catch (error) {
      console.error('Failed to record usage:', error);
      throw error;
    }
  }

  /**
   * Batch record usage events
   */
  async recordUsageBatch(usageRecords: UsageRecord[]): Promise<void> {
    const query = `
      INSERT INTO usage_records 
      (subscription_id, metric_name, quantity, timestamp, idempotency_key)
      VALUES ${usageRecords
        .map((_, i) => `($${i * 5 + 1}, $${i * 5 + 2}, $${i * 5 + 3}, $${i * 5 + 4}, $${i * 5 + 5})`)
        .join(',')}
      ON CONFLICT (idempotency_key) DO NOTHING
    `;

    const values = usageRecords.flatMap((u) => [
      u.subscriptionId,
      u.metricName,
      u.quantity,
      u.timestamp,
      this.generateIdempotencyKey(u),
    ]);

    await this.db.query(query, values);
  }

  /**
   * Get current usage for a subscription period
   */
  async getCurrentUsage(subscriptionId: string, metricName: string): Promise<number> {
    const query = `
      SELECT COALESCE(SUM(quantity), 0) as total_quantity
      FROM usage_records
      WHERE subscription_id = $1
        AND metric_name = $2
        AND timestamp >= CURRENT_DATE
        AND timestamp < CURRENT_DATE + INTERVAL '1 day'
    `;

    const result = await this.db.query(query, [subscriptionId, metricName]);
    return result.rows[0]?.total_quantity || 0;
  }

  /**
   * Calculate overage charges for a subscription
   */
  async calculateOverageCharges(subscriptionId: string): Promise<number> {
    const query = `
      SELECT COALESCE(SUM(pt.price_per_unit_cents * ur.quantity), 0) as overage_charge
      FROM usage_records ur
      JOIN subscriptions s ON ur.subscription_id = s.id
      JOIN pricing_plans p ON s.plan_id = p.id
      JOIN plan_tiers pt ON p.id = pt.plan_id
        AND ur.quantity BETWEEN pt.usage_start_units AND COALESCE(pt.usage_end_units, 999999999)
      WHERE ur.subscription_id = $1
        AND ur.timestamp >= s.current_period_start
        AND ur.timestamp < s.current_period_end
    `;

    const result = await this.db.query(query, [subscriptionId]);
    return result.rows[0]?.overage_charge || 0;
  }

  private generateIdempotencyKey(usage: UsageRecord): string {
    return `${usage.subscriptionId}:${usage.metricName}:${usage.timestamp.toISOString()}`;
  }
}

export default MeteringService;
```

```typescript
  });
```

```typescript
// Multi-tenancy isolation middleware
import { Request, Response, NextFunction } from 'express';

interface AuthRequest extends Request {
  tenantId?: string;
  userId?: string;
}

async function tenantIsolationMiddleware(
  req: AuthRequest,
  res: Response,
  next: NextFunction
) {
  const token = req.headers.authorization?.replace('Bearer ', '');

  if (!token) {
    return res.status(401).json({ error: 'Unauthorized' });
  }

  try {
    // Verify JWT and extract tenant ID
    const decoded = verifyJWT(token);
    req.tenantId = decoded.tenantId;
    req.userId = decoded.userId;

    // Add tenant context to all queries
    req.app.locals.currentTenantId = decoded.tenantId;

    next();
  } catch (error) {
    res.status(401).json({ error: 'Invalid token' });
  }
}

// Row-level security policy
async function enforceRowLevelSecurity(
  req: AuthRequest,
  res: Response,
  next: NextFunction
) {
  const { billingAccountId } = req.params;

  // Verify the billing account belongs to the tenant
  const result = await db.query(
    `SELECT * FROM billing_accounts 
     WHERE id = $1 AND tenant_id = $2`,
    [billingAccountId, req.tenantId]
  );

  if (result.rows.length === 0) {
    return res.status(403).json({ error: 'Forbidden' });
  }

  next();
}
```

## 9. Critical Implementation Tips

1. **Idempotency**: Always use idempotency keys for payment and usage recording
2. **Partitioning**: Partition usage_records table by date to handle high volume
3. **Caching**: Cache current usage and credit amounts in Redis
4. **Event-Driven**: Use event bus for asynchronous operations
5. **Rate Limiting**: Implement per-tenant rate limits on API endpoints
6. **Webhooks**: Send webhook notifications for important billing events
7. **Testing**: Comprehensive tests for subscription lifecycle and invoicing
8. **Documentation**: OpenAPI/Swagger spec for all endpoints

This architecture is production-ready and scales to handle millions of subscriptions while maintaining strict data isolation and compliance requirements.
