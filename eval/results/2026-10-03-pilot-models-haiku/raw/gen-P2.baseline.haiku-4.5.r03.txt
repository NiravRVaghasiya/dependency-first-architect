  case 'volume':\n        return this.calculateVolumeAmount(quantity, pricing.tiers);\n      \n      default:\n        throw new Error(`Unknown pricing model: ${pricing.model}`);\n    }\n  }\n\n  // PRORATION & PRORATING LOGIC\n\n### **Handling Plan Changes**\n\n```javascript\nclass SubscriptionService {\n  async changePlan(tenantId, subscriptionId, newPlanId, prorationBehavior) {\n    const subscription = await this.repo.get(tenantId, subscriptionId);\n    const currentPlan = await this.planRepo.get(tenantId, subscription.plan_id);\n    const newPlan = await this.planRepo.get(tenantId, newPlanId);\n    \n    const now = Date.now();\n    const periodEnd = subscription.current_period_end;\n    const daysRemaining = Math.ceil((periodEnd - now) / (24 * 60 * 60 * 1000));\n    const totalDays = Math.ceil(\n      (periodEnd - subscription.current_period_start) / (24 * 60 * 60 * 1000)\n    );\n    \n    const prorationDetails = {\n      current_plan_id: subscription.plan_id,\n      new_plan_id: newPlanId,\n      effective_date: new Date(now),\n      proration_date: new Date(now)\n    };\n    \n    switch (prorationBehavior) {\n      case 'create_prorations':\n        // Calculate pro-rata credit/charge\n        prorationDetails.proration = this.calculateProration(\n          currentPlan.amount,\n          newPlan.amount,\n          daysRemaining,\n          totalDays\n        );\n        \n        // Create invoice for the difference\n        if (prorationDetails.proration.amount !== 0) {\n          await this.createProratedInvoice(\n            tenantId,\n            subscription,\n            prorationDetails\n          );\n        }\n        break;\n      \n      case 'always_invoice':\n        // Charge full amount immediately, credit old subscription\n        prorationDetails.proration = {\n          credit_amount: this.calculateCredit(\n            currentPlan.amount,\n            daysRemaining,\n            totalDays\n          ),\n          new_charge: newPlan.amount,\n          net_amount: newPlan.amount - this.calculateCredit(\n            currentPlan.amount,\n            daysRemaining,\n            totalDays\n          )\n        };\n        \n        await this.createProratedInvoice(\n          tenantId,\n          subscription,\n          prorationDetails\n        );\n        break;\n      \n      case 'none':\n        // No proration, change at next renewal\n        subscription.pending_plan_id = newPlanId;\n        prorationDetails.effective_date = periodEnd;\n        break;\n    }\n    \n    // Update subscription\n    await this.repo.update(tenantId, subscriptionId, {\n      plan_id: prorationBehavior !== 'none' ? newPlanId : subscription.plan_id,\n      pending_plan_id: prorationBehavior === 'none' ? newPlanId : null,\n      ...prorationDetails\n    });\n    \n    // Publish event\n    await this.eventBus.publish('subscription.updated', {\n      subscription_id: subscriptionId,\n      ...prorationDetails\n    });\n  }\n\n  calculateProration(currentAmount, newAmount, daysRemaining, totalDays) {\n    const dailyRateCurrent = currentAmount / totalDays;\n    const creditsRemaining = dailyRateCurrent * daysRemaining;\n    \n    const dailyRateNew = newAmount / totalDays;\n    const chargeForRemainingDays = dailyRateNew * daysRemaining;\n    \n    const amount = chargeForRemainingDays - creditsRemaining;\n    \n    return {\n      credit_amount: creditsRemaining,\n      new_charge: chargeForRemainingDays,\n      amount,\n      reason: amount > 0 ? 'upgrade' : 'downgrade'\n    };\n  }\n}\n```\n\n## 10.
