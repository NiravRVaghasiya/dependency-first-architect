### 6.1 Subscription Service (Node.js/Python)

```python
# subscription_service.py
from datetime import datetime, timedelta
from enum import Enum
import uuid

class SubscriptionStatus(Enum):
    TRIAL = "trial"
    ACTIVE = "active"
    PAST_DUE = "past_due"
    CANCELLED = "cancelled"

class SubscriptionService:
    def __init__(self, db, payment_gateway, event_bus):
        self.db = db
        self.payment_gateway = payment_gateway
        self.event_bus = event_bus

    def create_subscription(self, tenant_id, plan_id, payment_method_id):
        """Create new subscription with trial period"""
        plan = self.db.query("plans").get(plan_id)
        
        # Calculate periods
        now = datetime.utcnow()
        trial_end = now + timedelta(days=plan.trial_days) if plan.trial_days else now
        period_end = now + timedelta(days=30)  # Simplified
        
        subscription = {
            "id": str(uuid.uuid4()),
            "tenant_id": tenant_id,
            "plan_id": plan_id,
            "status": SubscriptionStatus.TRIAL.value if plan.trial_days else SubscriptionStatus.ACTIVE.value,
            "current_period_start": now.isoformat(),
            "current_period_end": period_end.isoformat(),
            "billing_cycle_anchor": now.isoformat(),
            "created_at": now.isoformat()
        }
        
        self.db.insert("subscriptions", subscription)
        self.event_bus.emit("subscription.created", subscription)
        
        return subscription

    def upgrade_subscription(self, subscription_id, new_plan_id):
        """Handle plan upgrade with proration"""
        subscription = self.db.query("subscriptions").get(subscription_id)
        old_plan = self.db.query("plans").get(subscription["plan_id"])
        new_plan = self.db.query("plans").get(new_plan_id)
        
        # Calculate proration credit
        days_remaining = self._calculate_days_remaining(subscription)
        old_daily_rate = old_plan["price"] / 30
        new_daily_rate = new_plan["price"] / 30
        credit = old_daily_rate * days_remaining
        charge = (new_daily_rate * days_remaining) - credit
        
        # Update subscription
        self.db.update("subscriptions", subscription_id, {
            "plan_id": new_plan_id,
            "updated_at": datetime.utcnow().isoformat()
        })
        
        self.event_bus.emit("subscription.upgraded", {
            "subscription_id": subscription_id,
            "old_plan_id": old_plan["id"],
            "new_plan_id": new_plan_id,
            "proration_credit": credit,
            "proration_charge": charge
        })

    def cancel_subscription(self, subscription_id, immediately=False):
        """Cancel subscription"""
        if immediately:
            status = SubscriptionStatus.CANCELLED.value
            cancelled_at = datetime.utcnow().isoformat()
        else:
            status = SubscriptionStatus.ACTIVE.value
            cancelled_at = None
        
        self.db.update("subscriptions", subscription_id, {
            "status": status,
            "cancel_at_period_end": not immediately,
            "cancelled_at": cancelled_at
        })
        
        self.event_bus.emit("subscription.cancelled", {
            "subscription_id": subscription_id,
            "immediately": immediately
        })

    def _calculate_days_remaining(self, subscription):
        end = datetime.fromisoformat(subscription["current_period_end"])
        now = datetime.utcnow()
        return max((end - now).days, 0)
```

### 6.2 Billing Engine (Python)

```python
# billing_engine.py
from decimal import Decimal
from datetime import datetime

class BillingEngine:
    def __init__(self, db, usage_service, tax_service):
        self.db = db
        self.usage_service = usage_service
        self.tax_service = tax_service

    def generate_invoice(self, subscription_id):
        """Generate invoice for billing period"""
        subscription = self.db.query("subscriptions").get(subscription_id)
        tenant_id = subscription["tenant_id"]
        plan = self.db.query("plans").get(subscription["plan_id"])
        
        period_start = subscription["current_period_start"]
        period_end = subscription["current_period_end"]
        
        # Calculate charges
        line_items = []
        subtotal = Decimal('0')
        
        # 1. Base subscription charge
        line_items.append({
            "description": plan["name"],
            "quantity": 1,
            "unit_price": Decimal(str(plan["price"])),
            "amount": Decimal(str(plan["price"])),
            "line_type": "subscription"
        })
        subtotal += Decimal(str(plan["price"]))
        
        # 2. Usage-based charges
        usage_charges = self._calculate_usage_charges(tenant_id, period_start, period_end)
        for charge in usage_charges:
            line_items.append(charge)
            subtotal += charge["amount"]
        
        # 3. Calculate tax
        tax_amount = self.tax_service.calculate_tax(
            tenant_id, 
            subtotal,
            period_start
        )
        
        # 4. Apply discounts (if any)
        discount = self._calculate_discounts(tenant_id, subtotal)
        
        total = subtotal + tax_amount - discount
        
        # Create invoice
        invoice = {
            "id": str(uuid.uuid4()),
            "tenant_id": tenant_id,
            "subscription_id": subscription_id,
            "invoice_number": self._generate_invoice_number(tenant_id),
            "status": "draft",
            "period_start": period_start,
            "period_end": period_end,
            "due_date": self._calculate_due_date(period_end),
            "subtotal": float(subtotal),
            "tax": float(tax_amount),
            "discount": float(discount),
            "total": float(total),
            "currency": plan["currency"],
            "created_at": datetime.utcnow().isoformat(),
            "line_items": line_items
        }
        
        return self.db.insert("invoices", invoice)

    def _calculate_usage_charges(self, tenant_id, period_start, period_end):
        """Query usage and calculate charges"""
        charges = []
        
        # Get usage metrics for period
        usage_data = self.usage_service.get_usage(
            tenant_id, 
            period_start, 
            period_end
        )
        
        for metric_type, volume in usage_data.items():
            # Look up pricing for this metric
            pricing = self.db.query(
                "usage_pricing"
            ).filter(
                metric_type=metric_type
            ).first()
            
            if pricing:
                amount = Decimal(str(volume)) * Decimal(str(pricing["price_per_unit"]))
                charges.append({
                    "description": f"{metric_type} ({volume} units)",
                    "quantity": volume,
                    "unit_price": pricing["price_per_unit"],
                    "amount": float(amount),
                    "line_type": "usage"
                })
        
        return charges

    def finalize_invoice(self, invoice_id):
        """Finalize and send invoice"""
        invoice = self.db.query("invoices").get(invoice_id)
        
        # Generate PDF
        pdf_url = self._generate_pdf(invoice)
        
        # Update status
        self.db.update("invoices", invoice_id, {
            "status": "sent",
            "pdf_url": pdf_url,
            "sent_at": datetime.utcnow().isoformat()
        })
        
        # Send email
        self._send_invoice_email(invoice, pdf_url)
        
        return invoice
```

### 6.3 Metering Service (Go/Rust)

```go
// metering_service.go
package metering

import (
    "context"
    "time"
    "kafka"
    "timescaledb"
)

type UsageEvent struct {
    TenantID      string                 `json:"tenant_id"`
    MetricType    string                 `json:"metric_type"`
    Value         float64                `json:"value"`
    Timestamp     time.Time              `json:"timestamp"`
    IdempotencyKey string                `json:"idempotency_key"`
    Metadata      map[string]interface{} `json:"metadata"`
}

type MeteringService struct {
    kafkaProducer *kafka.Producer
    db            *timescaledb.DB
    cache         *redis.Client
}

func (ms *MeteringService) RecordEvent(ctx context.Context, event UsageEvent) error {
    // Validate
    if event.TenantID == "" || event.MetricType == "" {
        return ErrInvalidEvent
    }
    
    // Check idempotency
    cacheKey := "metering:" + event.IdempotencyKey
    if ms.cache.Exists(ctx, cacheKey).Val() == 1 {
        return nil // Already processed
    }
    
    // Publish to Kafka for async processing
    err := ms.kafkaProducer.Produce(
        &kafka.Message{
            Topic: "billing-events",
            Value: event,
        }, nil)
    
    // Mark as processed
    ms.cache.Set(ctx, cacheKey, "1", 24*time.Hour)
    
    return err
}

func (ms *MeteringService) GetAggregatedUsage(ctx context.Context, 
    tenantID string, startTime, endTime time.Time) (map[string]float64, error) {
    
    query := `
    SELECT metric_type, SUM(value) as total
    FROM tenant_usage
    WHERE tenant_id = $1 
    AND time >= $2 AND time < $3
    GROUP BY metric_type
    `
    
    rows, err := ms.db.Query(ctx, query, tenantID, startTime, endTime)
    if err != nil {
        return nil, err
    }
    
    usage := make(map[string]float64)
    for rows.Next() {
        var metricType string
        var total float64
        if err := rows.Scan(&metricType, &total); err != nil {
            return nil, err
        }
        usage[metricType] = total
    }
    
    return usage, nil
}
```
