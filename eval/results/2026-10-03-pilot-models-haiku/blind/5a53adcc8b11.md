**Rationale:**
- **Data isolation**: Each tenant's data completely isolated
- **Scalability**: Distribute tenants across database instances
- **Cost optimization**: Pay-as-you-grow database resources
- **Performance**: No cross-tenant query interference
- **Compliance**: GDPR/data residency requirements

```javascript
class BillingCycleStartedHandler {
  async handle(event) {
    const { subscription_id, period_start, period_end } = event;
    
    // Get subscription and aggregate data
    const subscription = await subscriptionRepo.get(subscription_id);
    const usageData = await usageService.aggregate({

      period_end

    const lineItems = [

        type: 'subscription',
        amount: subscription.plan.amount

    // Create invoice

      subscription_id,
      line_items: lineItems,
      amount: total,
      due_date: this.calculateDueDate(period_end)
    });
    
    // Publish event
    await eventBus.publish('invoice.created', { invoice_id: invoice.id });
  }
}

// Payment Success Handler
class PaymentSucceededHandler {
  async handle(event) {
    const { transaction_id, invoice_id } = event;
    
    // Mark invoice as paid
    await invoiceService.markPaid({
      invoice_id,
      transaction_id,
      paid_at: new Date()
    });
    
    // Mark subscription as up-to-date
    await subscriptionService.updateStatus({
      subscription_id: invoice.subscription_id,
      status: 'active'
    });

    await notificationService.send({
      customer_id: invoice.customer_id,
      type: 'payment_received',
      invoice_id
    });
  }

// Payment Failed Handler (Dunning)
class PaymentFailedHandler {
  async handle(event) {
    const { transaction_id, invoice_id } = event;
    
    const transaction = await transactionRepo.get(transaction_id);
    const invoice = await invoiceRepo.get(invoice_id);
    
    // Update retry count
    const retryCount = (invoice.retry_count || 0) + 1;
    await invoiceService.updateRetryCount(invoice_id, retryCount);
    
    // Determine next action
    if (retryCount < 3) {
      // Schedule retry
      const nextRetry = this.calculateNextRetry(retryCount);
      await schedulerService.schedule({
        type: 'retry_payment',
        invoice_id,
        scheduled_for: nextRetry
      });
      
      // Send reminder
      await notificationService.send({
        customer_id: invoice.customer_id,
        type: 'payment_failed_reminder',
        invoice_id,
        retry_count: retryCount
      });
    } else {
      // Suspend subscription after 3 failed attempts
      await subscriptionService.suspend(invoice.subscription_id);
      
      await notificationService.send({
        customer_id: invoice.customer_id,
        type: 'subscription_suspended',
        reason: 'payment_failed'
      });
    }
  }
}
```

## 8. USAGE METERING & AGGREGATION

### **Real-time Usage Tracking**

```javascript
// Metering Service Implementation
class MeteringService {
  // Store usage with idempotency
  async recordUsage(tenantId, {
    customer_id,
    meter_id,
    quantity,
    timestamp,
    idempotency_key
  }) {
    // Check for duplicate
    const existing = await this.usageRepo.findByIdempotencyKey(
      tenantId,
      idempotency_key
    );
    
    if (existing) {
      return existing; // Idempotent
    }
    
    // Record usage
    const usage = {
      usage_record_id: uuid(),
      customer_id,
      meter_id,
      quantity,
      timestamp: timestamp || Date.now(),
      idempotency_key,
      status: 'pending'
    };
    
    await this.usageRepo.save(tenantId, usage);
    
    // Publish event for async processing
    await this.eventBus.publish('usage.recorded', usage);
    
    // Update real-time counters (Redis)
    await this.updateRealTimeCounters(tenantId, customer_id, meter_id, quantity);
    
    return usage;
  }

  // Aggregate usage for current billing period
  async aggregateUsage(tenantId, customerId, meterId, period) {
    const subscription = await this.subscriptionRepo.getActive(
      tenantId,
      customerId
    );
    
    if (!subscription) {
      throw new Error('No active subscription');
    }
    
    const { current_period_start, current_period_end } = subscription;
    
    // Query usage records
    const records = await this.usageRepo.query(tenantId, {
      customer_id: customerId,
      meter_id: meterId,
      timestamp: {
        $gte: current_period_start,
        $lt: current_period_end
      }
    });
    
    const totalQuantity = records.reduce((sum, r) => sum + r.quantity, 0);
    
    // Get pricing model
    const plan = await this.planRepo.get(tenantId, subscription.plan_id);
    const meter = plan.meters.find(m => m.meter_id === meterId);
    
    // Calculate cost based on pricing model
    const amount = this.calculateAmount(totalQuantity, meter.pricing);
    
    return {
      meter_id: meterId,
      period_start: current_period_start,
      period_end: current_period_end,
      total_quantity: totalQuantity,
      unit_name: meter.unit_name,
      amount,
      detailed_breakdown: this.getDetailedBreakdown(totalQuantity, meter.pricing)
    };
  }

  // Calculate cost based on pricing model
  calculateAmount(quantity, pricing) {
    switch (pricing.model) {
      case 'per_unit':
        return quantity * pricing.unit_price;
      
      case 'tiered':
        return this.calculateTieredAmount(quantity, pricing.tiers);
      
      case 'volume':
        return this.calculateVolumeAmount(quantity, pricing.tiers);
      
      default:
        throw new Error(`Unknown pricing model: ${pricing.model}`);
    }
  }

  // Tiered: Price per unit decreases at higher tiers
  calculateTieredAmount(quantity, tiers) {
    let amount = 0;
    let remaining = quantity;
    
    for (const tier of tiers) {
      if (remaining <= 0) break;
      
      const tierSize = tier.to ? tier.to - tier.from : remaining;
      const tierQuantity = Math.min(remaining, tierSize);
      
      amount += tierQuantity * tier.unit_price;
      remaining -= tierQuantity;
    }
    
    return amount;
  }

  // Volume: Entire volume gets the price of the tier it falls into
  calculateVolumeAmount(quantity, tiers) {
```

```javascript
class SubscriptionService {
  async changePlan(tenantId, subscriptionId, newPlanId, prorationBehavior) {
    const subscription = await this.repo.get(tenantId, subscriptionId);
    const currentPlan = await this.planRepo.get(tenantId, subscription.plan_id);
    const newPlan = await this.planRepo.get(tenantId, newPlanId);
    
    const now = Date.now();
    const periodEnd = subscription.current_period_end;
    const daysRemaining = Math.ceil((periodEnd - now) / (24 * 60 * 60 * 1000));
    const totalDays = Math.ceil(
      (periodEnd - subscription.current_period_start) / (24 * 60 * 60 * 1000)
    );
    
    const prorationDetails = {
      current_plan_id: subscription.plan_id,
      new_plan_id: newPlanId,
      effective_date: new Date(now),
      proration_date: new Date(now)
    };
    
    switch (prorationBehavior) {
      case 'create_prorations':
        // Calculate pro-rata credit/charge
        prorationDetails.proration = this.calculateProration(
          currentPlan.amount,
          newPlan.amount,
          daysRemaining,
          totalDays
        );
        
        // Create invoice for the difference
        if (prorationDetails.proration.amount !== 0) {
          await this.createProratedInvoice(
            tenantId,
            subscription,
            prorationDetails
          );
        }
        break;
      
      case 'always_invoice':
        // Charge full amount immediately, credit old subscription
        prorationDetails.proration = {
          credit_amount: this.calculateCredit(
            currentPlan.amount,
            daysRemaining,
            totalDays
          ),
          new_charge: newPlan.amount,
          net_amount: newPlan.amount - this.calculateCredit(
            currentPlan.amount,
            daysRemaining,
            totalDays
          )
        };
        
        await this.createProratedInvoice(
          tenantId,
          subscription,
          prorationDetails
        );
        break;
      
      case 'none':
        // No proration, change at next renewal
        subscription.pending_plan_id = newPlanId;
        prorationDetails.effective_date = periodEnd;
        break;
    }
    
    // Update subscription
    await this.repo.update(tenantId, subscriptionId, {
      plan_id: prorationBehavior !== 'none' ? newPlanId : subscription.plan_id,
      pending_plan_id: prorationBehavior === 'none' ? newPlanId : null,
      ...prorationDetails
    });
    
    // Publish event
    await this.eventBus.publish('subscription.updated', {
      subscription_id: subscriptionId,
      ...prorationDetails
    });
  }

  calculateProration(currentAmount, newAmount, daysRemaining, totalDays) {
    const dailyRateCurrent = currentAmount / totalDays;
    const creditsRemaining = dailyRateCurrent * daysRemaining;
    
    const dailyRateNew = newAmount / totalDays;
    const chargeForRemainingDays = dailyRateNew * daysRemaining;
    
    const amount = chargeForRemainingDays - creditsRemaining;
    
    return {
      credit_amount: creditsRemaining,
      new_charge: chargeForRemainingDays,
      amount,
      reason: amount > 0 ? 'upgrade' : 'downgrade'
    };
  }
}
```

```javascript
class PaymentService {
  constructor(gatewayFactory) {
    this.gatewayFactory = gatewayFactory;
  }

  async processPayment(tenantId, invoiceId, paymentMethodId) {
    const invoice = await this.invoiceRepo.get(tenantId, invoiceId);
    const paymentMethod = await this.paymentMethodRepo.get(
      tenantId,
      paymentMethodId
    );
    const customer = await this.customerRepo.get(
      tenantId,
      invoice.customer_id
    );
    
    // Create transaction record
    const transaction = {
      transaction_id: uuid(),
      invoice_id: invoiceId,
      payment_method_id: paymentMethodId,
      amount: invoice.amount,
      currency: invoice.currency,
      status: 'pending',
      created_at: Date.now()
    };
    
    await this.transactionRepo.save(tenantId, transaction);
    
    try {
      // Get appropriate payment gateway
      const gateway = this.gatewayFactory.create(
        paymentMethod.provider,
        tenantId
      );
      
      // Process payment
      const result = await gateway.charge({
        amount: invoice.amount,
        currency: invoice.currency,
        payment_method_id: paymentMethod.provider_reference_id,
        customer_id: customer.provider_customer_id,
        idempotency_key: transaction.transaction_id,
        description: `Invoice ${invoiceId}`,
        metadata: {
          invoice_id: invoiceId,
          tenant_id: tenantId,
          customer_id: customer.customer_id
        }
      });
      
      // Update transaction
      transaction.status = 'success';
      transaction.provider_transaction_id = result.transaction_id;
      transaction.processed_at = Date.now();
      
      await this.transactionRepo.update(tenantId, transaction.transaction_id, transaction);
      
      // Publish event
      await this.eventBus.publish('payment.succeeded', {
        transaction_id: transaction.transaction_id,
        invoice_id: invoiceId,
        amount: invoice.amount
      });
      
      return transaction;
      
    } catch (error) {
      // Update transaction
      transaction.status = 'failed';
      transaction.error_message = error.message;
      transaction.error_code = error.code;
      
      await this.transactionRepo.update(tenantId, transaction.transaction_id, transaction);
      
      // Publish event
      await this.eventBus.publish('payment.failed', {
        transaction_id: transaction.transaction_id,
        invoice_id: invoiceId,
        error_code: error.code,
        error_message: error.message
      });
      
      throw error;
    }
  }

  // Retry failed payment with exponential backoff
  async retryPayment(tenantId, invoiceId, attempt = 1) {
    const maxAttempts = 3;
    const backoffMs = 60000 * Math.pow(2, attempt - 1); // 1m, 2m, 4m
    
    if (attempt > maxAttempts) {
      throw new Error('Max retry attempts exceeded');
    }
    
    try {
      await this.processPayment(
        tenantId,
        invoiceId,
        invoice.payment_method_id
      );
    } catch (error) {
      // Schedule next retry
      await this.schedulerService.scheduleIn({
        type: 'retry_payment',
        invoice_id: invoiceId,
        attempt: attempt + 1,
        delay: backoffMs
      });
      
      throw error;
    }
  }

  // Handle webhooks from payment providers
  async handleWebhook(tenantId, provider, signature, payload) {
    // Verify signature
    const gateway = this.gatewayFactory.create(provider, tenantId);
    const verified = await gateway.verifyWebhookSignature(signature, payload);
    
    if (!verified) {
      throw new Error('Invalid webhook signature');
    }
    
    const event = gateway.parseWebhookPayload(payload);
    
    // Process based on event type
    switch (event.type) {
      case 'charge.succeeded':
        await this.handleChargeSucceeded(tenantId, event.data);
        break;
      
      case 'charge.failed':
        await this.handleChargeFailed(tenantId, event.data);
        break;
      
      case 'charge.refunded':
        await this.handleChargeRefunded(tenantId, event.data);
        break;
      
      default:
        console.log(`Unhandled webhook event: ${event.type}`);
    }
  }

  async handleChargeSucceeded(tenantId, data) {
    const transaction = await this.transactionRepo.findByProvider(
      tenantId,
      data.provider_transaction_id
    );
    
    transaction.status = 'success';
    transaction.processed_at = Date.now();
    
    await this.transactionRepo.update(tenantId, transaction.transaction_id, transaction);
    await this.eventBus.publish('payment.succeeded', transaction);
  }

  async handleChargeFailed(tenantId, data) {
    const transaction = await this.transactionRepo.findByProvider(
      tenantId,
      data.provider_transaction_id
    );
    
    transaction.status = 'failed';
    transaction.error_code = data.error_code;
    transaction.error_message = data.error_message;
    
    await this.transactionRepo.update(tenantId, transaction.transaction_id, transaction);
    await this.eventBus.publish('payment.failed', transaction);
  }
}
```

```javascript
class InvoiceService {

      subscriptionId
    );
    const customer = await this.customerRepo.get(
      tenantId,
      subscription.customer_id
    );

    // Build line items
    const lineItems = [];
    
    // 1. Subscription line item
    lineItems.push({
      type: 'subscription',
      description: plan.name,
      quantity: 1,
      unit_price: plan.amount,
      amount: plan.amount,
      period_start: periodStart,
      period_end: periodEnd
    });
    
    // 2. Usage-based line items
    if (plan.meters && plan.meters.length > 0) {
      for (const meter of plan.meters) {
        const usage = await this.meteringService.aggregateUsage(
          tenantId,
          customer.customer_id,
          meter.meter_id,
          { start: periodStart, end: periodEnd }
        );
        
        if (usage.amount > 0) {
          lineItems.push({
            type: 'usage',
            description: `${meter.name} (${usage.total_quantity} ${meter.unit_name})`,
            quantity: usage.total_quantity,
            unit_price: 0, // Varies by tier
            amount: usage.amount,
            breakdown: usage.detailed_breakdown
          });
        }
      }
    }
    
    // 3. Discounts/Credits
    const credits = await this.creditRepo.query(tenantId, {
      customer_id: customer.customer_id,
      status: 'available'
    });
    
    let totalBeforeCredits = lineItems.reduce((sum, li) => sum + li.amount, 0);
    
    for (const credit of credits) {
      if (credit.amount >= totalBeforeCredits) {
        lineItems.push({
          type: 'credit',
          description: credit.description,
          amount: -totalBeforeCredits
        });
        totalBeforeCredits = 0;
        break;
      } else {
        lineItems.push({
          type: 'credit',
          description: credit.description,
          amount: -credit.amount
        });
        totalBeforeCredits -= credit.amount;
      }
    }
    
    // Calculate totals
    const subtotal = lineItems
      .filter(li => li.type !== 'tax')
      .reduce((sum, li) => sum + li.amount, 0);
    
    // 4. Tax
    let tax = 0;
    if (customer.tax_id) {
      tax = await this.taxService.calculateTax({
        customer_tax_id: customer.tax_id,
        amount: subtotal
      });
      
      lineItems.push({
        type: 'tax',
        description: 'Sales Tax',
        amount: tax
      });
    }
    
    const total = subtotal + tax;
    
    // Create invoice
    const invoice = {
      invoice_id: `inv_${uuid()}`,
      tenant_id: tenantId,
      customer_id: customer.customer_id,
      subscription_id: subscriptionId,
      line_items: lineItems,
      subtotal,
      tax,
      total,
      currency: plan.currency,
      status: 'draft',
      issue_date: new Date(),
      due_date: this.calculateDueDate(new Date(), 30),
      period_start: periodStart,
      period_end: periodEnd
    };
    
    await this.invoiceRepo.save(tenantId, invoice);
    
    // Publish event
    await this.eventBus.publish('invoice.created', {
      invoice_id: invoice.invoice_id,
      customer_id: customer.customer_id,
      amount: invoice.total
    });
    
    return invoice;
  }
```

```javascript
// API Security Middleware
class SecurityMiddleware {
  authenticate(req, res, next) {
    const token = req.headers.authorization?.split(' ')[1];
    
    if (!token) {
      return res.status(401).json({ error: 'Missing authorization token' });
    }
    
    try {
      const decoded = jwt.verify(token, process.env.JWT_SECRET, {
        algorithms: ['HS256'],
        issuer: 'billing-system',
        audience: 'api-clients'
      });
      
      req.tenantId = decoded.tenant_id;
      req.userId = decoded.user_id;
      req.scopes = decoded.scopes;
      
      next();
    } catch (error) {
      return res.status(401).json({ error: 'Invalid token' });
    }
  }

  authorize(requiredScopes) {
    return (req, res, next) => {
      const hasScopes = requiredScopes.every(scope => 
        req.scopes.includes(scope)
      );
      
      if (!hasScopes) {
        return res.status(403).json({ error: 'Insufficient permissions' });
      }
      
      next();
    };
  }

  validateTenantAccess(req, res, next) {
    const pathTenantId = req.params.tenantId || req.query.tenantId;
    
    if (pathTenantId && pathTenantId !== req.tenantId) {
      return res.status(403).json({ error: 'Tenant access denied' });
    }
    
    next();
  }

  rateLimiting(req, res, next) {
    const key = `ratelimit:${req.tenantId}:${req.path}`;
    const limit = 100; // requests per minute
    
    redis.incr(key, (err, count) => {
      if (err) throw err;
      
      if (count === 1) {
        redis.expire(key, 60);
      }
      
      res.setHeader('X-RateLimit-Limit', limit);
      res.setHeader('X-RateLimit-Remaining', Math.max(0, limit - count));
      
      if (count > limit) {
        return res.status(429).json({ error: 'Rate limit exceeded' });
      }
      
      next();
    });
  }

  requestSigning(req, res, next) {
    const signature = req.headers['x-signature'];
    const timestamp = req.headers['x-timestamp'];
    const body = req.rawBody; // Must capture before parsing
    
    if (!signature || !timestamp) {
      return res.status(401).json({ error: 'Missing signature' });
    }
    
    // Prevent replay attacks
    if (Math.abs(Date.now() - parseInt(timestamp)) > 300000) { // 5 min
      return res.status(401).json({ error: 'Request expired' });
    }
    
    // Verify HMAC
    const hmac = crypto
      .createHmac('sha256', process.env.API_SECRET)
      .update(`${timestamp}.${body}`)
      .digest('hex');
    
    if (hmac !== signature) {
      return res.status(401).json({ error: 'Invalid signature' });
    }
    
    next();
  }

  auditLogging(req, res, next) {
    const startTime = Date.now();
    
    res.on('finish', () => {
      const duration = Date.now() - startTime;
      
      auditLogger.log({
        timestamp: new Date(),
        tenant_id: req.tenantId,
        user_id: req.userId,
        method: req.method,
        path: req.path,
        status_code: res.statusCode,
        duration_ms: duration,
        ip_address: req.ip,
        user_agent: req.headers['user-agent'],
        request_id: req.headers['x-request-id']
      });
    });
    
    next();
  }
}

// Encryption for sensitive fields
class FieldEncryption {
  constructor(tenantId) {
    this.tenantId = tenantId;
    this.key = this.deriveKey(tenantId);
  }

  deriveKey(tenantId) {
    return crypto
      .pbkdf2Sync(
        process.env.ENCRYPTION_MASTER_KEY + tenantId,
        process.env.SALT,
        100000,
        32,
        'sha256'
      );
  }

  encrypt(plaintext) {
    const iv = crypto.randomBytes(16);
    const cipher = crypto.createCipheriv('aes-256-gcm', this.key, iv);
    
    let encrypted = cipher.update(plaintext, 'utf8', 'hex');
    encrypted += cipher.final('hex');
    
    const authTag = cipher.getAuthTag();
    
    return `${iv.toString('hex')}:${authTag.toString('hex')}:${encrypted}`;
  }

  decrypt(encrypted) {
    const [iv, authTag, ciphertext] = encrypted.split(':');
    
    const decipher = crypto.createDecipheriv(
      'aes-256-gcm',
      this.key,
      Buffer.from(iv, 'hex')
    );
    
    decipher.setAuthTag(Buffer.from(authTag, 'hex'));
    
    let decrypted = decipher.update(ciphertext, 'hex', 'utf8');
    decrypted += decipher.final('utf8');
    
    return decrypted;
  }
}

class PIIMasking {
  static mask(value, type) {
    switch (type) {
      case 'email':
        return value.replace(/(.{2})(.*)(@.*)/, '$1***$3');
      
      case 'credit_card':
        return value.replace(/(\d{4})(\d{8})(\d{4})/, '$1-****-****-$3');
      
      case 'phone':
        return value.replace(/(\d{3})(\d{3})(\d{4})/, '$1-***-$3');
      
      case 'ssn':
        return value.replace(/(\d{3})-?(\d{2})-?(\d{4})/, '***-**-$3');
      
      default:
        return '***';
    }
  }
```

## **Summary: Key Design Principles**

1. **Multi-tenancy**: Database per tenant for isolation and scalability
2. **Flexibility**: Support multiple billing models and pricing strategies
3. **Reliability**: Event-driven architecture with idempotency and retries
4. **Performance**: Real-time usage tracking with Redis caching
5. **Security**: Encryption, audit logging, compliance-ready
6. **Scalability**: Horizontal scaling with Kubernetes, microservices
7. **Observability**: Comprehensive logging, monitoring, and alerting
8. **Extensibility**: Plugin architecture for payment gateways and integrations

This architecture is production-ready and handles enterprise-scale requirements while maintaining developer experience and operational simplicity.
