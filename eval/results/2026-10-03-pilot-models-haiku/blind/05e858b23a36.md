## 4. Billing Engine Architecture

```typescript
// Billing Engine - Core Logic
class BillingEngine {
  private eventBus: EventBus;
  private repository: BillingRepository;
  private paymentProcessor: PaymentProcessor;

  // Monthly billing cycle execution
  async processBillingCycle(
    billingPeriod: BillingPeriod
  ): Promise<Invoice[]> {
    const invoices: Invoice[] = [];

    // 1. Fetch all active subscriptions for period
    const subscriptions = await this.repository.getActiveSubscriptions(
      billingPeriod.tenantId,
      billingPeriod
    );

    for (const subscription of subscriptions) {
      // 2. Aggregate usage metrics
      const usage = await this.aggregateUsage(subscription, billingPeriod);

      // 3. Calculate charges
      const charges = await this.calculateCharges(subscription, usage);

      // 4. Apply credits and adjustments
      const adjustedCharges = await this.applyAdjustments(
        charges,
        subscription
      );

      // 5. Apply taxes
      const withTax = await this.calculateTax(adjustedCharges, subscription);

      // 6. Generate invoice
      const invoice = await this.createInvoice(
        subscription,
        billingPeriod,
        withTax
      );

      invoices.push(invoice);

      // 7. Emit event for downstream processing
      await this.eventBus.publish('invoice.created', {
        invoiceId: invoice.id,
        tenantId: invoice.tenantId,
        amount: invoice.total
      });
    }

    return invoices;
  }

  private async aggregateUsage(
    subscription: Subscription,
    period: BillingPeriod
  ): Promise<UsageData> {
    // Aggregate metered usage during billing period
    const meterReadings = await this.repository.getMeterReadings(
      subscription.id,
      period.startDate,
      period.endDate
    );

    return {
      features: subscription.features.map(feature => ({
        id: feature.id,
        usage: meterReadings
          .filter(m => m.featureId === feature.id)
          .reduce((sum, m) => sum + m.usage, 0),
        overage: 0
      }))
    };
  }

  private async calculateCharges(
    subscription: Subscription,
    usage: UsageData
  ): Promise<LineItem[]> {
    const charges: LineItem[] = [];

    // 1. Base subscription charge
    const baseCharge = this.calculateBaseCharge(subscription);
    charges.push(baseCharge);

    // 2. Usage-based charges
    for (const feature of subscription.features) {
      const usageItem = usage.features.find(u => u.id === feature.id);
      if (usageItem) {
        if (feature.type === 'metered') {
          const charge = this.calculateMeteredCharge(feature, usageItem.usage);
          if (charge.amount > 0) charges.push(charge);
        } else if (feature.type === 'tiered') {
          const charge = this.calculateTieredCharge(feature, usageItem.usage);
          if (charge.amount > 0) charges.push(charge);
        }
      }
    }

    // 3. Add-on charges
    for (const addOn of subscription.addOns) {
      const addOnCharge = this.calculateAddOnCharge(addOn);
      charges.push(addOnCharge);
    }

    return charges;
  }

  private async applyAdjustments(
    charges: LineItem[],
    subscription: Subscription
  ): Promise<LineItem[]> {
    // Apply credits, discounts, promotions
    const credits = await this.repository.getActiveCredits(subscription.id);
    const discounts = await this.repository.getActiveDiscounts(
      subscription.id
    );

    let items = [...charges];

    for (const credit of credits) {
      items.push({
        type: 'credit',
        description: credit.reason,
        amount: -credit.amount,
        quantity: 1,
        unitPrice: { amount: -credit.amount, currency: 'USD' },
        subtotal: { amount: -credit.amount, currency: 'USD' }
      });
    }

    for (const discount of discounts) {
      const subtotal = items
        .filter(i => i.type !== 'credit')
        .reduce((sum, i) => sum + i.subtotal.amount, 0);
      const discountAmount = subtotal * (discount.percentage / 100);
      items.push({
        type: 'adjustment',
        description: discount.name,
        amount: -discountAmount,
        quantity: 1,
        unitPrice: { amount: -discountAmount, currency: 'USD' },
        subtotal: { amount: -discountAmount, currency: 'USD' }
      });
    }

    return items;
  }

  private async calculateTax(
    charges: LineItem[],
    subscription: Subscription
  ): Promise<LineItem[]> {
    // Calculate taxes based on tenant location and tax configuration
    const subtotal = charges.reduce((sum, item) => sum + item.subtotal.amount, 0);
    
    const taxCalculation = await this.taxService.calculateTax({
      tenantId: subscription.tenantId,
      amount: subtotal
    });

    return [
      ...charges,
      {
        type: 'adjustment',
        description: 'Tax',
        amount: taxCalculation.amount,
        quantity: 1,
        unitPrice: { amount: taxCalculation.amount, currency: 'USD' },
        subtotal: { amount: taxCalculation.amount, currency: 'USD' }
      }
    ];
  }

  private calculateBaseCharge(subscription: Subscription): LineItem {
    const plan = subscription.plan;
    return {
      type: 'subscription',
      description: `${plan.name} - Monthly`,
      quantity: 1,
      unitPrice: plan.basePrice,
      subtotal: plan.basePrice,
      id: generateId()
    };
  }

  private calculateMeteredCharge(
    feature: Feature,
    usage: number
  ): LineItem {
    const limit = feature.limits;
    if (usage <= limit.limit) {
      return {
        type: 'usage',
        description: feature.name,
        quantity: usage,
        unitPrice: limit.overage.pricePerUnit,
        subtotal: {
          amount: usage * limit.overage.pricePerUnit.amount,
          currency: limit.overage.pricePerUnit.currency
        },
        id: generateId()
      };
    }
    return null;
  }

  private calculateTieredCharge(
    feature: Feature,
    usage: number
  ): LineItem {
    // Tier 1: 0-1000 @ $0.10
    // Tier 2: 1001-5000 @ $0.08
    // Tier 3: 5001+ @ $0.05
    let charge = 0;
    // Logic for tiered pricing calculation
    return {
      type: 'usage',
      description: `${feature.name} (tiered)`,
      quantity: usage,
      unitPrice: { amount: charge / usage, currency: 'USD' },
      subtotal: { amount: charge, currency: 'USD' },
      id: generateId()
    };
  }

  private calculateAddOnCharge(addOn: AddOn): LineItem {
    return {
      type: 'addon',
      description: addOn.name,
      quantity: addOn.quantity,
      unitPrice: addOn.price,
      subtotal: {
        amount: addOn.quantity * addOn.price.amount,
        currency: addOn.price.currency
      },
      id: generateId()
    };
  }

  private async createInvoice(
    subscription: Subscription,
    billingPeriod: BillingPeriod,
    lineItems: LineItem[]
  ): Promise<Invoice> {
    const subtotal = lineItems
      .filter(item => item.type !== 'adjustment')
      .reduce((sum, item) => sum + item.subtotal.amount, 0);

    const tax = lineItems
      .filter(item => item.description === 'Tax')
      .reduce((sum, item) => sum + item.amount, 0);

    const total = lineItems.reduce((sum, item) => sum + item.subtotal.amount, 0);

    const invoice: Invoice = {
      id: generateId(),
      tenantId: subscription.tenantId,
      subscriptionId: subscription.id,
      billingPeriodId: billingPeriod.id,
      lineItems,
      subtotal: { amount: subtotal, currency: 'USD' },
      tax: { amount: tax, currency: 'USD' },
      total: { amount: total, currency: 'USD' },
      dueDate: new Date(Date.now() + 30 * 24 * 60 * 60 * 1000),
      status: 'draft',
      createdAt: new Date()
    };

    await this.repository.createInvoice(invoice);
    return invoice;
  }
}
```

## 5. Usage Metering Service

```typescript
class MeteringService {
  private eventStore: EventStore;
  private aggregationEngine: AggregationEngine;
  private cache: Cache;

  // Record usage event (high throughput)
  async recordUsageEvent(event: UsageEvent): Promise<void> {
    // 1. Validate event
    await this.validateUsageEvent(event);

    // 2. Store event (immutable)
    await this.eventStore.append(event);

    // 3. Update aggregations asynchronously
    await this.eventBus.publish('usage.recorded', event);

    // 4. Invalidate cache
    await this.cache.invalidate(`usage:${event.subscriptionId}`);
  }

  // Batch record usage (for high-volume integrations)
  async recordBatchUsage(events: UsageEvent[]): Promise<void> {
    // 1. Validate all events
    const validatedEvents = await Promise.all(
      events.map(e => this.validateUsageEvent(e))
    );

    // 2. Store in bulk
    await this.eventStore.appendBatch(validatedEvents);

    // 3. Trigger async processing
    for (const event of validatedEvents) {
      await this.eventBus.publish('usage.recorded', event);
    }
  }

  // Get usage for billing period
  async getUsageForPeriod(
    subscriptionId: string,
    startDate: Date,
    endDate: Date
  ): Promise<FeatureUsage[]> {
    const cacheKey = `usage:${subscriptionId}:${startDate}:${endDate}`;
    
    // Check cache first
    const cached = await this.cache.get(cacheKey);
    if (cached) return cached;

    // Aggregate from event store
    const events = await this.eventStore.query({
      subscriptionId,
      timestamp: { $gte: startDate, $lte: endDate }
    });

    const usage = this.aggregateEvents(events);
    
    // Cache result
    await this.cache.set(cacheKey, usage, { ttl: 3600 });

    return usage;
  }

  // Real-time usage dashboard
  async getCurrentPeriodUsage(subscriptionId: string): Promise<CurrentUsage> {
    const subscription = await this.getSubscription(subscriptionId);
    const usage = await this.getUsageForPeriod(
      subscriptionId,
      subscription.currentPeriodStart,
      subscription.currentPeriodEnd
    );

    return {
      subscriptionId,
      period: {
        start: subscription.currentPeriodStart,
        end: subscription.currentPeriodEnd
      },
      features: usage,
      percentageUsed: usage.map(f => ({
        featureId: f.featureId,
        used: f.quantity,
        limit: f.limit,
        percentage: (f.quantity / f.limit) * 100
      }))
    };
  }

  private aggregateEvents(events: UsageEvent[]): FeatureUsage[] {
    const grouped = new Map<string, UsageEvent[]>();

    for (const event of events) {
      const key = event.featureId;
      if (!grouped.has(key)) {
        grouped.set(key, []);
      }
      grouped.get(key)!.push(event);
    }

    return Array.from(grouped.entries()).map(([featureId, featureEvents]) => ({
      featureId,
      quantity: featureEvents.reduce((sum, e) => sum + e.quantity, 0),
      eventCount: featureEvents.length,
      lastUpdated: new Date(Math.max(...featureEvents.map(e => e.timestamp.getTime())))
    }));
  }

  private async validateUsageEvent(event: UsageEvent): Promise<UsageEvent> {
    // Validate subscription is active
    const subscription = await this.getSubscription(event.subscriptionId);
    if (subscription.status !== 'active') {
      throw new Error('Subscription is not active');
    }

    // Validate feature exists and is within plan
    const feature = subscription.features.find(f => f.id === event.featureId);
    if (!feature) {
      throw new Error('Feature not found in subscription');
    }

    // Validate usage quantity
    if (event.quantity < 0) {
      throw new Error('Usage quantity cannot be negative');
    }

    return event;
  }
}
```

## 6. Payment Processing Integration

```typescript
class PaymentProcessor {
  private paymentGateway: PaymentGateway; // Stripe, Adyen, etc.
  private eventBus: EventBus;
  private repository: BillingRepository;

  async processPayment(
    invoice: Invoice,
    paymentMethod: PaymentMethod
  ): Promise<Payment> {
    try {
      // 1. Create payment record (pending)
      const payment = await this.repository.createPayment({
        invoiceId: invoice.id,
        tenantId: invoice.tenantId,
        amount: invoice.total,
        method: paymentMethod.type,
        status: 'pending'
      });

      // 2. Call payment gateway
      const gatewayResponse = await this.paymentGateway.charge({
        amount: invoice.total.amount,
        currency: invoice.total.currency,
        customerId: await this.getTenantCustomerId(invoice.tenantId),
        paymentMethodId: paymentMethod.id,
        idempotencyKey: payment.id, // Prevent duplicate charges
        metadata: {
          invoiceId: invoice.id,
          subscriptionId: invoice.subscriptionId,
          tenantId: invoice.tenantId
        }
      });

      // 3. Update payment status
      const updatedPayment = await this.repository.updatePayment(payment.id, {
        status: 'succeeded',
        transactionId: gatewayResponse.transactionId,
        processedAt: new Date()
      });

      // 4. Update invoice status
      await this.repository.updateInvoice(invoice.id, {
        status: 'paid',
        paidAt: new Date()
      });

      // 5. Publish event
      await this.eventBus.publish('payment.succeeded', {
        paymentId: payment.id,
        invoiceId: invoice.id,
        tenantId: invoice.tenantId,
        amount: invoice.total
      });

      return updatedPayment;

    } catch (error) {
      // 1. Update payment status to failed
      await this.repository.updatePayment(payment.id, {
        status: 'failed',
        errorMessage: error.message
      });

      // 2. Publish event
      await this.eventBus.publish('payment.failed', {
        paymentId: payment.id,
        invoiceId: invoice.id,
        tenantId: invoice.tenantId,
        reason: error.message
      });

      // 3. Trigger retry logic or escalation
      await this.handleFailedPayment(invoice);

      throw error;
    }
  }

  private async handleFailedPayment(invoice: Invoice): Promise<void> {
    // 1. Retry with exponential backoff
    const retryCount = await this.repository.getPaymentRetryCount(invoice.id);
    if (retryCount < 3) {
      const delayMs = Math.pow(2, retryCount) * 3600000; // 1h, 2h, 4h
      await this.scheduleRetry(invoice.id, delayMs);
    } else {
      // 2. Notify tenant
      await this.notificationService.sendPaymentFailedAlert(invoice);
      
      // 3. Escalate
      await this.escallateToSupport(invoice);
    }
  }

  // Handle webhook callbacks from payment gateway
  async handlePaymentWebhook(webhookData: WebhookEvent): Promise<void> {
    const { type, data } = webhookData;

    switch (type) {
      case 'charge.succeeded':
        await this.handleChargeSucceeded(data);
        break;
      case 'charge.failed':
        await this.handleChargeFailed(data);
        break;
      case 'charge.refunded':
        await this.handleChargeRefunded(data);
        break;
      case 'customer.subscription.deleted':
        await this.handleSubscriptionCanceled(data);
        break;
    }
  }

  private async handleChargeSucceeded(data: any): Promise<void> {
    const payment = await this.repository.getPaymentByTransactionId(
      data.id
    );
    if (!payment) return;

    await this.repository.updatePayment(payment.id, {
      status: 'succeeded',
      processedAt: new Date()
    });

    // Verify invoice is marked as paid
    await this.repository.updateInvoice(payment.invoiceId, {
      status: 'paid',
      paidAt: new Date()
    });
  }

  private async handleChargeFailed(data: any): Promise<void> {
    const payment = await this.repository.getPaymentByTransactionId(data.id);
    if (!payment) return;

    await this.repository.updatePayment(payment.id, {
      status: 'failed',
      errorMessage: data.failure_message
    });

    await this.handleFailedPayment(
      await this.repository.getInvoice(payment.invoiceId)
    );
  }

  private async handleChargeRefunded(data: any): Promise<void> {
    const payment = await this.repository.getPaymentByTransactionId(data.id);
    if (!payment) return;

    await this.repository.updatePayment(payment.id, {
      status: 'refunded'
    });

    await this.eventBus.publish('payment.refunded', {
      paymentId: payment.id,
      invoiceId: payment.invoiceId,
      amount: data.amount_refunded
    });
  }
}
```

```typescript
// Idempotency for safe retries
class IdempotencyManager {
  async executeWithIdempotency<T>(
    key: string,
    operation: () => Promise<T>
  ): Promise<T> {
    // Check if operation already executed
    const cached = await this.cache.get(`idempotency:${key}`);
    if (cached) {
      return cached;
    }

    // Execute operation
    const result = await operation();

    // Cache result
    await this.cache.set(`idempotency:${key}`, result, { ttl: 24 * 3600 });

    return result;
  }
}

// Event sourcing for audit trail
class EventSourcingStore {
  async appendEvent(event: BillingEvent): Promise<void> {
    // Immutable log of all billing events
    await this.eventLog.append({
      id: generateId(),
      type: event.type,
      aggregateId: event.aggregateId,
      tenantId: event.tenantId,
      data: event.data,
      timestamp: new Date(),
      version: await this.getNextVersion(event.aggregateId)
    });
  }

  async getEventHistory(
    aggregateId: string,
    fromVersion: number = 0
  ): Promise<BillingEvent[]> {
    return await this.eventLog.query({
      aggregateId,
      version: { $gte: fromVersion }
    });
  }

  // Replay events to rebuild state
  async rebuildAggregateState(aggregateId: string): Promise<any> {
    const events = await this.getEventHistory(aggregateId);
    let state = {};

    for (const event of events) {
      state = this.applyEvent(state, event);
    }

    return state;
  }
}

// Caching strategy
class CachingStrategy {
  // Cache invalidation on state changes
  async updateSubscription(id: string, update: Partial<Subscription>) {
    await this.repository.update(id, update);
    
    // Invalidate related caches
    await this.cache.invalidate(`subscription:${id}`);
    await this.cache.invalidate(`usage:${id}:*`); // Pattern invalidation
    await this.cache.invalidate(`invoice:subscription:${id}`);
  }

  // Time-based cache with refresh
  async getCachedSubscription(id: string): Promise<Subscription> {
    const key = `subscription:${id}`;
    let subscription = await this.cache.get(key);

    if (!subscription) {
      subscription = await this.repository.get(id);
      await this.cache.set(key, subscription, { ttl: 300 }); // 5 minutes
    }

    return subscription;
  }
}
```

```typescript
class BillingTransaction {
  private saga: Saga;

  async executeInvoiceAndPaymentFlow(invoice: Invoice): Promise<void> {
    const saga = new Saga('invoice-payment-flow');

    // Step 1: Create invoice
    saga.addStep({
      name: 'create-invoice',
      action: () => this.createInvoice(invoice),
      compensation: () => this.deleteInvoice(invoice.id)
    });

    // Step 2: Charge payment
    saga.addStep({
      name: 'charge-payment',
      action: () => this.chargePayment(invoice),
      compensation: () => this.refundPayment(invoice.id)
    });

      action: () => this.sendInvoiceNotification(invoice),
      compensation: () => {} // No compensation needed
    });

    // Step 4: Update subscription renewal date
    saga.addStep({
      name: 'update-subscription',
      action: () => this.updateSubscriptionRenewalDate(invoice.subscriptionId),
      compensation: () => this.revertSubscriptionRenewalDate(invoice.subscriptionId)
    });

    await saga.execute();
  }

// Duplicate prevention with deduplication tokens
class DeduplicationService {
  async recordWithDeduplication(
    tenantId: string,
    deduplicationKey: string,
    data: any
  ): Promise<any> {
    const hash = sha256(`${tenantId}:${deduplicationKey}`);
    const existing = await this.cache.get(`dedup:${hash}`);

    if (existing) {
      return existing;
    }

    const result = await this.processData(data);
    await this.cache.set(`dedup:${hash}`, result, { ttl: 86400 }); // 24h

    return result;
  }
}

// Circuit breaker for payment gateway
class PaymentGatewayCircuitBreaker {
  private state: 'closed' | 'open' | 'half-open' = 'closed';
  private failureCount = 0;
  private successCount = 0;

  async call<T>(operation: () => Promise<T>): Promise<T> {
    if (this.state === 'open') {
      throw new Error('Circuit breaker is OPEN - service unavailable');
    }

    try {
      const result = await operation();
      this.onSuccess();
      return result;
    } catch (error) {
      this.onFailure();
      throw error;
    }
  }

  private onSuccess(): void {
    this.failureCount = 0;
    if (this.state === 'half-open') {
      this.state = 'closed';
      this.successCount = 0;
    }
  }

  private onFailure(): void {
    this.failureCount++;
    if (this.failureCount >= 5) {
      this.state = 'open';
      setTimeout(() => {
        this.state = 'half-open';
        this.failureCount = 0;
      }, 60000); // 1 minute
    }
  }
}
```

## Key Design Principles

1. **Idempotency** - All operations can be safely retried
2. **Event Sourcing** - Complete audit trail of all billing events
3. **CQRS** - Separate read and write models for scalability
4. **Saga Pattern** - Distributed transaction management
5. **Circuit Breaker** - Graceful degradation
6. **Tenant Isolation** - Security and data privacy
7. **Immutability** - Usage events and audit logs
8. **Caching** - Performance optimization
9. **Monitoring** - Comprehensive observability
10. **Async Processing** - High throughput metering

This architecture supports millions of transactions, handles complex pricing models, and maintains complete audit trails for compliance.
